normalize to Facility if "Facility" not in df.columns and "Facility_Name" in df.columns: df["Facility"] = df["Facility_Name"] # Backfill County/SubCounty/Facility if missing if "County" not in df.columns: df["County"] = "Unknown" if "SubCounty" not in df.columns: df["SubCounty"] = "Unknown" if "Facility" not in df.columns: df["Facility"] = "Unknown" if "Sex" not in df.columns: df["Sex"] = "Unknown" if "AgeBand" not in df.columns: # compute from Age at Reporting if "Age at Reporting" in df.columns: df["AgeBand"] = df["Age at Reporting"].apply(make_ageband) else: df["AgeBand"] = "Unknown" # Regimen Line normalization if "Current Regimen Line" not in df.columns: df["Current Regimen Line"] = "Unknown" # Regimen Type normalization (prefer existing Regimen_Type if present) if "Regimen_Type" not in df.columns: if "Current Regimen" in df.columns: df["Regimen_Type"] = df["Current Regimen"].apply(classify_regimen_type) else: df["Regimen_Type"] = "Other" # Sex normalization df["Sex_norm"] = df["Sex"].apply(norm_sex) # Reporting Month display (string) if "Reporting Month" in df.columns: # allow date or text def format_rm(x): if pd.isna(x): return "" if isinstance(x, (pd.Timestamp, datetime)): return x.strftime("%b%Y") # e.g., Dec2025 s = str(x).strip() # if already Dec2025, keep; if 2025-12-31, try parse try: dt = pd.to_datetime(s, errors="coerce") if not pd.isna(dt): return dt.strftime("%b%Y") except: pass return s df["ReportingMonthText"] = df["Reporting Month"].apply(format_rm) else: df["ReportingMonthText"] = "" # Build row-level payload (browser does filtering + aggregation) rows = [] for _, r in df.iterrows(): rows.append({ "County": safe_str(r.get("County", "Unknown")) or "Unknown", "SubCounty": safe_str(r.get("SubCounty", "Unknown")) or "Unknown", "Facility": safe_str(r.get("Facility", "Unknown")) or "Unknown", "Sex": safe_str(r.get("Sex_norm", "Unknown")) or "Unknown", "AgeBand": safe_str(r.get("AgeBand", "Unknown")) or "Unknown", "Regimen_Line": safe_str(r.get("Current Regimen Line", "Unknown")) or "Unknown", "Regimen_Type": safe_str(r.get("Regimen_Type", "Other")) or "Other", "ReportingMonth": safe_str(r.get("ReportingMonthText", "")) or "" }) # Metadata meta = { "generated": datetime.now().strftime("%Y-%m-%d %H:%M"), "reporting_month": (df["ReportingMonthText"].dropna().iloc[0] if "ReportingMonthText" in df.columns and not df["ReportingMonthText"].dropna().empty else ""), "rows": int(len(rows)) } data_json = json.dumps({"meta": meta, "rows": rows}, ensure_ascii=False) # Inject JSON into template with open(TEMPLATE_HTML, "r", encoding="utf-8") as f: html = f.read() if "__DATA_JSON__" not in html: print("❌ Template missing placeholder __DATA_JSON__") return out_html = html.replace("__DATA_JSON__", data_json) with open(OUTPUT_HTML, "w", encoding="utf-8") as f: f.write(out_html) print("✅ ART Optimization dashboard generated") print(f"📄 Output: {OUTPUT_HTML}") print(f"📦 Rows embedded: {len(rows):,}") print("=" * 100) if __name__ == "__main__": main()
